Prisma's `where` delegates line up almost one-to-one with the grid's filter operators, so
the mapping is a table, not a switch:

```ts
// app/api/grid/route.ts
import { Prisma } from "@prisma/client";
import { prisma } from "~/lib/prisma";

// grid operator → Prisma delegate; the "" pair is the empty/notEmpty values
const OPS: Record<string, string> = {
  contains: "contains",
  notContains: "notContains",
  equals: "equals",
  notEquals: "notEquals",
  startsWith: "startsWith",
  endsWith: "endsWith",
  gt: "gt",
  gte: "gte",
  lt: "lt",
  lte: "lte",
  empty: "equals",
  notEmpty: "notEquals",
};

const ALLOWED = new Set(["name", "email", "role", "score"]);

// The body is untrusted: cap its size at the framework or proxy, and bound the tree before it reaches a query.
function assertFilterTree(nodes, depth = 0, count = { leaves: 0, groups: 0 }) {
  if (!Array.isArray(nodes)) throw new Error("invalid filters");
  for (const node of nodes) {
    if (Array.isArray(node?.filters)) {
      if (depth >= 2 || ++count.groups > 16 || (node.join !== "and" && node.join !== "or")) throw new Error("invalid group");
      assertFilterTree(node.filters, depth + 1, count);
    } else if (!ALLOWED.has(node?.columnId) || !Object.hasOwn(OPS, node.operator) || ++count.leaves > 64) {
      throw new Error("invalid filter");
    }
  }
}

function toWhere(node): Prisma.UserWhereInput {
  if (Array.isArray(node.filters)) {
    if (node.filters.length === 0) return {};
    const children = node.filters.map(toWhere);
    return node.join === "or" ? { OR: children } : { AND: children };
  }
  const value = node.operator === "empty" ? "" : node.operator === "notEmpty" ? { not: "" } : node.value;
  return { [node.columnId]: { [OPS[node.operator]]: value } };
}

export async function POST(req: Request) {
  const { page, pageSize, sorts, filters, search } = await req.json();
  assertFilterTree(filters);

  const where: Prisma.UserWhereInput = {
    ...(search
      ? { OR: [{ name: { contains: search, mode: "insensitive" } }, { email: { contains: search, mode: "insensitive" } }] }
      : {}),
    ...(filters.length ? { AND: filters.map(toWhere) } : {}),
  };

  const [total, rows] = await Promise.all([
    prisma.user.count({ where }),
    prisma.user.findMany({
      where,
      orderBy: sorts.length
        ? sorts.map((s) => ({ [s.columnId]: s.direction }))
        : [{ id: "asc" }],
      skip: (page - 1) * pageSize,
      take: pageSize,
    }),
  ]);

  return Response.json({ rows, total });
}
```

Notes:

- `mode: "insensitive"` gives the case-insensitive `contains` the grid's quick search
  implies. Drop it for exact semantics.
- `isAnyOf` maps to `in: f.value`, and `isNoneOf` maps to `notIn: f.value`. Add them to the table above when you need them.
- The client is the [server-side example](/docs/examples/data/server-side) unchanged.
  Edits become `prisma.user.update` in the same write-through handler.
